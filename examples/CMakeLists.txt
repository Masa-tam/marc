cmake_minimum_required(VERSION 3.25)

project(marc_c_roundtrip_example LANGUAGES C CXX)

find_package(marc CONFIG REQUIRED)

# Optional package-validation assertion; normal consumers need not set it.
set(MARC_EXPECTED_LINKAGE "" CACHE STRING "Expected package targets: shared, static, or both")
if(NOT MARC_EXPECTED_LINKAGE STREQUAL "")
    if(NOT MARC_EXPECTED_LINKAGE MATCHES "^(shared|static|both)$")
        message(FATAL_ERROR "Invalid MARC_EXPECTED_LINKAGE: ${MARC_EXPECTED_LINKAGE}")
    endif()
    foreach(kind IN ITEMS shared static)
        if(MARC_EXPECTED_LINKAGE STREQUAL "both" OR MARC_EXPECTED_LINKAGE STREQUAL kind)
            if(NOT TARGET marc::${kind})
                message(FATAL_ERROR "Expected package target marc::${kind} is missing")
            endif()
        elseif(TARGET marc::${kind})
            message(FATAL_ERROR "Unexpected package target marc::${kind}")
        endif()
    endforeach()
endif()

enable_testing()

add_executable(marc_c_roundtrip c_roundtrip.c)
target_compile_features(marc_c_roundtrip PRIVATE c_std_11)
if(TARGET marc::shared)
    target_link_libraries(marc_c_roundtrip PRIVATE marc::shared)
elseif(TARGET marc::static)
    target_link_libraries(marc_c_roundtrip PRIVATE marc::static)
    set_property(TARGET marc_c_roundtrip PROPERTY LINKER_LANGUAGE CXX)
else()
    message(FATAL_ERROR "The marc package exports no supported library target")
endif()

add_test(NAME baseline_roundtrip COMMAND marc_c_roundtrip)
foreach(kind IN ITEMS static shared)
    if(TARGET marc::${kind})
        add_executable(marc_position_distance_${kind} position_distance_roundtrip.c)
        target_compile_features(marc_position_distance_${kind} PRIVATE c_std_11)
        target_link_libraries(marc_position_distance_${kind} PRIVATE marc::${kind})
        if(kind STREQUAL "static")
            # The C ABI has a C++ implementation; keep sources C11, link as C++.
            set_property(TARGET marc_position_distance_${kind} PROPERTY LINKER_LANGUAGE CXX)
        endif()
        add_test(NAME position_distance_${kind} COMMAND marc_position_distance_${kind})
    endif()
endforeach()
